\item A circuit consists of a battery of emf $\varepsilon = 12 \ \mathrm{V}$ and internal resistance $r = 1 \ \Omega$ connected to two resistors $R_1 = 4 \ \Omega$ and $R_2 = 6 \ \Omega$ in parallel. The current drawn from the battery is

\begin{center}
    \input{diagram}
\end{center}

\begin{tasks}(2)
    \task $3 \ \mathrm{A}$
    \task $4 \ \mathrm{A}$
    \task $\frac{30}{17} \ \mathrm{A}$ \ans
    \task $5 \ \mathrm{A}$
\end{tasks}

\begin{solution}
\begin{align*}
\intertext{The equivalent resistance of $R_1$ and $R_2$ in parallel is}
R_{\text{eq}} &= \frac{R_1 R_2}{R_1 + R_2} \\
    &= \frac{4 \times 6}{4 + 6} \\
    &= \frac{24}{10} \\
    &= 2.4 \ \Omega
\intertext{Total resistance in the circuit}
R_{\text{total}} &= R_{\text{eq}} + r \\
    &= 2.4 + 1 \\
    &= 3.4 \ \Omega
\intertext{Current drawn from the battery}
I &= \frac{\varepsilon}{R_{\text{total}}} \\
    &= \frac{12}{3.4} \\
    &= \frac{30}{17} \ \mathrm{A}
\end{align*}
\end{solution}
