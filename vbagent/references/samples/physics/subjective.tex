\item A uniform rod of mass $m = 2 \ \mathrm{kg}$ and length $l = 1 \ \mathrm{m}$ is hinged at one end to a wall and held horizontal by a string attached to the other end making an angle of $30^\circ$ with the rod. Find the tension in the string and the reaction at the hinge.

    \input{diagram}
\end{center}

\begin{solution}
\begin{align*}
\intertext{Taking torques about the hinge point}
T \sin 30^\circ \cdot l &= mg \cdot \frac{l}{2} \\
T \cdot \frac{1}{2} \cdot 1 &= 2 \times 9.8 \times \frac{1}{2} \\
T &= \frac{2 \times 9.8}{1} \\
    &= 19.6 \ \mathrm{N}
\intertext{For equilibrium in the horizontal direction}
R_x &= T \cos 30^\circ \\
    &= 19.6 \times \frac{\sqrt{3}}{2} \\
    &= 9.8\sqrt{3} \ \mathrm{N}
\intertext{For equilibrium in the vertical direction}
R_y + T \sin 30^\circ &= mg \\
R_y &= mg - T \sin 30^\circ \\
    &= 19.6 - 19.6 \times \frac{1}{2} \\
    &= 9.8 \ \mathrm{N}
\intertext{The reaction at the hinge}
R &= \sqrt{R_x^2 + R_y^2} \\
    &= \sqrt{\left(9.8\sqrt{3}\right)^2 + 9.8^2} \\
    &= 9.8 \times 2 \\
    &= 19.6 \ \mathrm{N}
\end{align*}
\end{solution}
