\item Calculate the pH of a $0.1 \ \mathrm{M}$ solution of acetic acid ($\ce{CH3COOH}$) given that $K_a = 1.8 \times 10^{-5}$.

\begin{solution}
\begin{align*}
\intertext{For a weak acid $\ce{CH3COOH <=> CH3COO^- + H^+}$, the dissociation constant is}
K_a &= \frac{[\ce{CH3COO^-}][\ce{H^+}]}{[\ce{CH3COOH}]}
\intertext{Let $\alpha$ be the degree of dissociation. At equilibrium}
[\ce{H^+}] &= C\alpha = \sqrt{K_a \cdot C} \\
    &= \sqrt{1.8 \times 10^{-5} \times 0.1} \\
    &= \sqrt{1.8 \times 10^{-6}} \\
    &= 1.34 \times 10^{-3} \ \mathrm{M}
\intertext{Therefore}
\mathrm{pH} &= -\log[\ce{H^+}] \\
    &= -\log\left(1.34 \times 10^{-3}\right) \\
    &= 3 - \log 1.34 \\
    &= 3 - 0.127 \\
    &= 2.87
\end{align*}
\end{solution}
