\item Match Column I (organisms) with Column II (their characteristics).

\begin{center}
    \renewcommand{\arraystretch}{2}
    \begin{tabular}{@{}p{0.1\textwidth}p{0.3\textwidth}|p{0.1\textwidth}p{0.4\textwidth}@{}}
    \hline
    & \textbf{Column-I} & & \textbf{Column-II} \\
    \hline
    (a) & \textit{Plasmodium vivax} & (p) & Nitrogen fixation \\
    (b) & \textit{Rhizobium leguminosarum} & (q) & Causes malaria \\
    (c) & \textit{Lactobacillus acidophilus} & (r) & Curd formation \\
    (d) & \textit{Penicillium notatum} & (s) & Antibiotic production \\
    \hline
    \end{tabular}
\end{center}

\begin{tasks}(2)
    \task $\mathrm{a\rightarrow q,\ b\rightarrow p,\ c\rightarrow r,\ d\rightarrow s}$ \ans
    \task $\mathrm{a\rightarrow p,\ b\rightarrow q,\ c\rightarrow s,\ d\rightarrow r}$
    \task $\mathrm{a\rightarrow r,\ b\rightarrow s,\ c\rightarrow p,\ d\rightarrow q}$
    \task $\mathrm{a\rightarrow s,\ b\rightarrow r,\ c\rightarrow q,\ d\rightarrow p}$
\end{tasks}

\begin{solution}
    \begin{align*}
        \intertext{Analyze each organism:}
        \intertext{(a) \textit{Plasmodium vivax} is the causative agent of malaria $\rightarrow$ (q)}
        \intertext{(b) \textit{Rhizobium leguminosarum} is a nitrogen-fixing bacterium in root nodules of legumes $\rightarrow$ (p)}
        \intertext{(c) \textit{Lactobacillus acidophilus} is used in curd/yoghurt formation $\rightarrow$ (r)}
        \intertext{(d) \textit{Penicillium notatum} produces the antibiotic penicillin $\rightarrow$ (s)}
        \intertext{Therefore, the correct option is (a).}
    \end{align*}
\end{solution}
