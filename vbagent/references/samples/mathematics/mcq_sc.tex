\item The area bounded by the curve $y = x^2$, the $x$-axis, and the lines $x = 1$ and $x = 3$ is

\begin{center}
    \input{diagram}
\end{center}

\begin{tasks}(2)
    \task $\frac{26}{3}$ \ans
    \task $\frac{28}{3}$
    \task $9$
    \task $\frac{22}{3}$
\end{tasks}

\begin{solution}
\begin{align*}
A &= \int_{1}^{3} x^2 \, dx \\
    &= \left[\frac{x^3}{3}\right]_{1}^{3} \\
    &= \frac{27}{3} - \frac{1}{3} \\
    &= \frac{26}{3}
\end{align*}
\end{solution}
