\item Evaluate $\displaystyle\int_{0}^{\pi/2} \dfrac{\sin x}{\sin x + \cos x} \, dx$.

\begin{solution}
\begin{align*}
\intertext{Let}
I &= \int_{0}^{\pi/2} \dfrac{\sin x}{\sin x + \cos x} \, dx
\intertext{Using the property $\displaystyle\int_{0}^{a} f(x) \, dx = \int_{0}^{a} f(a - x) \, dx$ with $a = \dfrac{\pi}{2}$}
I &= \int_{0}^{\pi/2} \dfrac{\sin\left(\dfrac{\pi}{2} - x\right)}{\sin\left(\dfrac{\pi}{2} - x\right) + \cos\left(\dfrac{\pi}{2} - x\right)} \, dx \\
    &= \int_{0}^{\pi/2} \dfrac{\cos x}{\cos x + \sin x} \, dx
\intertext{Adding the two expressions for $I$}
2I &= \int_{0}^{\pi/2} \dfrac{\sin x + \cos x}{\sin x + \cos x} \, dx \\
    &= \int_{0}^{\pi/2} 1 \, dx \\
    &= \dfrac{\pi}{2}
\intertext{Therefore}
I &= \dfrac{\pi}{4}
\end{align*}
\end{solution}
