\begin{enumerate}[leftmargin=*]
\item A $2 \ \mathrm{A}$ current is flowing through two different small circular copper coils having radii ratio $1 : 2$. The ratio of their respective magnetic moments will be
\begin{tasks}(2)
    \task $2 : 1$
    \task $4 : 1$
    \task $1 : 4$ \ans
    \task $1 : 2$
\end{tasks}
\begin{solution}
    \begin{align*}
    m &= I A\\
    A &= \pi r^2\\
    \therefore m &\propto r^2\\
    \intertext{Since current is same in both coils and $r_1 : r_2 = 1 : 2$,}
    m_1 : m_2 &= r_1^2 : r_2^2\\
    &= 1^2 : 2^2\\
    &= 1 : 4\\
    \intertext{Therefore, the correct option is (c).}
    \end{align*}
\end{solution}

\begin{idea}
    \begin{align*}
    \intertext{\textbf{Concept:} Magnetic moment of a circular current loop}
    m &= IA \\
    A &= \pi r^2 \\
    \intertext{\textbf{Apply:} for two coils with the same current, $I_1=I_2$}
    \frac{m_1}{m_2} &= \frac{I_1A_1}{I_2A_2} \\
    &= \frac{A_1}{A_2} \\
    &= \frac{\pi r_1^2}{\pi r_2^2} \\
    &= \frac{r_1^2}{r_2^2} \\
    \intertext{\textbf{Technique:} Use $m \propto A$ and for a circular loop $A \propto r^2$, so $m \propto r^2$.}
    \end{align*}
\end{idea}
\end{enumerate}
