\begin{solution}
    \begin{align*}
    m &= I A\\
    A &= \pi r^2\\
    \therefore m &\propto r^2\\
    \intertext{Since current is same in both coils and $r_1 : r_2 = 1 : 2$,}
    m_1 : m_2 &= r_1^2 : r_2^2\\
    &= 1^2 : 2^2\\
    &= 1 : 4\\
    \intertext{Therefore, the correct option is (c).}
    \end{align*}
\end{solution}
